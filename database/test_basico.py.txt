def test_entrada_exitosa(app):
    """Prueba que una entrada se registre correctamente"""
    with app.app_context():
        user = Usuario.query.first()
        resultado = ServicioParqueadero.registrar_entrada("MKT456", user.id)
        assert "Entrada exitosa" in resultado
        assert Celda.query.filter_by(codigo="A-01", estado="Ocupada").first() is not None

def test_vehiculo_ya_dentro(app):
    """Prueba que no se permita ingresar un vehículo que ya está dentro"""
    with app.app_context():
        user = Usuario.query.first()
        ServicioParqueadero.registrar_entrada("MKT456", user.id)
        
        with pytest.raises(ReglaNegocioError, match="ya se encuentra dentro"):
            ServicioParqueadero.registrar_entrada("MKT456", user.id)

def test_sin_mensualidad(app):
    """Prueba que se rechace un vehículo sin contrato activo"""
    with app.app_context():
        user = Usuario.query.first()
        # Crear vehículo sin contrato
        v = Vehiculo(placa="SIN123", tipo="Moto", color="Negro", propietario="Test")
        db.session.add(v)
        db.session.commit()
        
        with pytest.raises(ReglaNegocioError, match="no tiene un contrato"):
            ServicioParqueadero.registrar_entrada("SIN123", user.id)
